public static class Server
{
	public static int Run(params IBackend[] backends) { /* as today, calling Serve(backends, conn) */ }

	public static void Serve(IReadOnlyList<IBackend> backends, Conn conn)
	{
		var hello = conn.Recv(Handshake.Parser) ?? throw new EndOfStreamException("...");
		var refusal = Incompatible(hello);
		IBackend? backend = null;
		if (refusal is null)
		{
			// Unverified: Environment.ProcessPath may resolve a symlink to its
			// target; GetCommandLineArgs()[0] should keep argv[0]. Check
			// under Native AOT before relying on either.
			(backend, refusal) = Choose(hello.Target, Environment.GetCommandLineArgs()[0], backends);
		}
		// ...then exactly today's Serve(IBackend, Conn) body for `backend`.
	}

	// Same rules as Go's choose: the handshake's target, then argv[0] without
	// "tdl-gen-", one backend serves anything when no target is named, and a
	// refusal names what was asked for and what is served.
	static (IBackend?, string?) Choose(string requested, string argv0, IReadOnlyList<IBackend> backends) { /* ... */ }
}
